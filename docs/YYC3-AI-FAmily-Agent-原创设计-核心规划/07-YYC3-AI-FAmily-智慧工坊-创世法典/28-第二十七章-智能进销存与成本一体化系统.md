38. private Map\<String, BigDecimal\> allocateCostsByABC(CostAllocationContext context) {

40. 

41. // 识别作业活动

42. List\<Activity\> activities = identifyActivities(context);

45. Map\<Activity, CostDriver\> costDrivers = determineCostDrivers(activities);

49.

51. for (CostObject costObject : context.getCostObjects()) {

52. BigDecimal objectCost = BigDecimal.ZERO;

53.

56. BigDecimal activityCost = activityRates.get(activity).multiply(driverQuantity);

57. objectCost = objectCost.add(activityCost);

58. }

59. 

60. allocatedCosts.put(costObject.getId(), objectCost);

61. }

62. 

63. return allocatedCosts;

64. }

42. private Map\<String, BigDecimal\> calculateSafetyStocks(DemandForecast forecast,

43. InventoryOptimizationContext context) {

44. Map\<String, BigDecimal\> safetyStocks = new HashMap\<\>();

45. 

46. for (String materialId : forecast.getMaterialDemands().keySet()) {

47. MaterialDemand demand = forecast.getMaterialDemands().get(materialId);

48. 

49. // 计算需求标准差

50. BigDecimal demandStd = calculateDemandStandardDeviation(demand);

51. 

52. // 获取提前期

53. LeadTime leadTime = getLeadTime(materialId, context);

54. 

55. // 计算提前期标准差

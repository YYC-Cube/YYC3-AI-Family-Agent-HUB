## 智枢服务化平台数据库

1.  -- 服务维度表

2.  CREATE TABLE service_dimensions (

3.  id BIGSERIAL PRIMARY KEY,

4.  dimension_code VARCHAR(50) NOT NULL UNIQUE,

5.  dimension_name VARCHAR(100) NOT NULL,

6.  description TEXT,

7.  capabilities JSONB NOT NULL,

8.  metrics_definition JSONB NOT NULL,

9.  status VARCHAR(20) DEFAULT 'ACTIVE',

10. created_by BIGINT NOT NULL,

11. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

12. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

13. );

14. 

15. -- 服务模板表

16. CREATE TABLE service_templates (

17. id BIGSERIAL PRIMARY KEY,

18. template_code VARCHAR(50) NOT NULL UNIQUE,

19. template_name VARCHAR(200) NOT NULL,

20. template_type VARCHAR(50) NOT NULL,

21. blueprint JSONB NOT NULL,

22. parameters_schema JSONB NOT NULL,

23. quality_standards JSONB NOT NULL,

24. adaptation_rules JSONB,

25. version VARCHAR(20) NOT NULL,

26. status VARCHAR(20) DEFAULT 'DRAFT',

27. created_by BIGINT NOT NULL,

28. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

29. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

30. );

31. 

32. -- 服务实例表

33. CREATE TABLE service_instances (

34. id BIGSERIAL PRIMARY KEY,

35. instance_code VARCHAR(50) NOT NULL UNIQUE,

36. template_id BIGINT NOT NULL REFERENCES service_templates(id),

37. tenant_id BIGINT NOT NULL,

38. configuration JSONB NOT NULL,

39. status VARCHAR(20) DEFAULT 'CREATING',

40. deployment_info JSONB,

41. quality_metrics JSONB,

42. performance_metrics JSONB,

43. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

44. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

45. );

46. 

47. -- 一致性记录表

48. CREATE TABLE consistency_records (

49. id BIGSERIAL PRIMARY KEY,

50. service_instance_id BIGINT NOT NULL REFERENCES service_instances(id),

51. consistency_type VARCHAR(50) NOT NULL,

52. check_point VARCHAR(100) NOT NULL,

53. expected_value JSONB,

54. actual_value JSONB,

55. deviation_level VARCHAR(20),

56. corrective_actions JSONB,

57. checked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

58. );

59. 

60. -- 价值度量表

61. CREATE TABLE value_metrics (

62. id BIGSERIAL PRIMARY KEY,

63. service_instance_id BIGINT NOT NULL REFERENCES service_instances(id),

64. metric_type VARCHAR(50) NOT NULL,

65. metric_name VARCHAR(100) NOT NULL,

66. metric_value DECIMAL(15,4) NOT NULL,

67. target_value DECIMAL(15,4),

68. unit VARCHAR(20) NOT NULL,

69. period_start TIMESTAMP NOT NULL,

70. period_end TIMESTAMP NOT NULL,

71. calculated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

72. );

73. 

74. -- 合作伙伴表

75. CREATE TABLE partners (

76. id BIGSERIAL PRIMARY KEY,

77. partner_code VARCHAR(50) NOT NULL UNIQUE,

78. partner_name VARCHAR(200) NOT NULL,

79. partner_type VARCHAR(50) NOT NULL,

80. capabilities JSONB NOT NULL,

81. service_offerings JSONB NOT NULL,

82. quality_rating DECIMAL(3,2) DEFAULT 5.0,

83. status VARCHAR(20) DEFAULT 'ACTIVE',

84. contracted_at TIMESTAMP,

85. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

86. );

87. 

88. -- 服务协同表

89. CREATE TABLE service_collaborations (

90. id BIGSERIAL PRIMARY KEY,

91. collaboration_code VARCHAR(50) NOT NULL UNIQUE,

92. initiator_id BIGINT NOT NULL,

93. participant_ids JSONB NOT NULL,

94. service_blueprint JSONB NOT NULL,

95. workflow_definition JSONB NOT NULL,

96. value_distribution JSONB NOT NULL,

97. status VARCHAR(20) DEFAULT 'ACTIVE',

98. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

99. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

100. );

### 数据库索引优化

1.  -- 性能关键索引

2.  CREATE INDEX idx_service_instances_tenant_status ON service_instances(tenant_id, status);

3.  CREATE INDEX idx_consistency_records_instance_type ON consistency_records(service_instance_id, consistency_type);

4.  CREATE INDEX idx_value_metrics_instance_period ON value_metrics(service_instance_id, period_start, period_end);

5.  CREATE INDEX idx_service_templates_type_status ON service_templates(template_type, status);

6.  CREATE INDEX idx_partners_type_rating ON partners(partner_type, quality_rating);

7.  

8.  -- 全文搜索索引

9.  CREATE INDEX idx_service_templates_blueprint_gin ON service_templates USING GIN (blueprint);

10. CREATE INDEX idx_service_instances_configuration_gin ON service_instances USING GIN (configuration);

11. 

12. -- 分区表设置

13. CREATE TABLE value_metrics_2024 PARTITION OF value_metrics

14. FOR VALUES FROM ('2024-01-01') TO ('2025-01-01');

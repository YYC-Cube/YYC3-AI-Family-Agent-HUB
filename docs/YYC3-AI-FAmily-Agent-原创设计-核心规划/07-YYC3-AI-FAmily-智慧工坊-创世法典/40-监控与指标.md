### 自定义指标

1.  @Component

2.  public class AIMetrics {

3.  

4.  private final MeterRegistry meterRegistry;

5.  

6.  // 对话相关指标

7.  private final Counter totalConversations;

8.  private final Counter failedConversations;

9.  private final Timer responseTimeTimer;

10. private final Gauge sentimentScoreGauge;

11. 

12. // 学习相关指标

13. private final Counter learningUpdates;

14. private final Gauge knowledgeBaseSize;

15. private final Gauge modelAccuracy;

16. 

17. public AIMetrics(MeterRegistry meterRegistry) {

18. this.meterRegistry = meterRegistry;

19. 

20. this.totalConversations = Counter.builder("ai.conversations.total")

21. .description("总对话数量")

22. .register(meterRegistry);

23. 

24. this.failedConversations = Counter.builder("ai.conversations.failed")

25. .description("失败对话数量")

26. .register(meterRegistry);

27. 

28. this.responseTimeTimer = Timer.builder("ai.response.time")

29. .description("响应时间分布")

30. .register(meterRegistry);

31. 

32. this.sentimentScoreGauge = Gauge.builder("ai.sentiment.score")

33. .description("情感分析得分")

34. .register(meterRegistry);

35. 

36. this.learningUpdates = Counter.builder("ai.learning.updates")

37. .description("学习更新次数")

38. .register(meterRegistry);

39. 

40. this.knowledgeBaseSize = Gauge.builder("ai.knowledge.size")

41. .description("知识库大小")

42. .register(meterRegistry);

43. 

44. this.modelAccuracy = Gauge.builder("ai.model.accuracy")

45. .description("模型准确率")

46. .register(meterRegistry);

47. }

48. 

49. public void recordConversation(boolean success, long responseTime, double sentimentScore) {

50. totalConversations.increment();

51. if (!success) {

52. failedConversations.increment();

53. }

54. responseTimeTimer.record(responseTime, TimeUnit.MILLISECONDS);

55. sentimentScoreGauge.set(sentimentScore);

56. }

57. 

58. public void recordLearningUpdate(int newKnowledgeItems) {

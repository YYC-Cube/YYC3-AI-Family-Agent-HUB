## 🗄️ 数据库设计扩展

### 新增数据表

1.  -- AI客服对话记录表

2.  CREATE TABLE ai_conversations (

3.  id BIGSERIAL PRIMARY KEY,

4.  session_id VARCHAR(100) NOT NULL,

5.  user_id BIGINT NOT NULL,

6.  mode VARCHAR(20) NOT NULL, -- 'customer' or 'operations'

7.  user_message TEXT NOT NULL,

8.  assistant_response TEXT NOT NULL,

9.  intent VARCHAR(100),

10. sentiment_score DECIMAL(3,2),

11. emotion_tags JSONB,

12. context_data JSONB,

13. actions_executed JSONB,

14. response_time_ms INTEGER,

15. user_rating INTEGER, -- 用户评分 1-5

16. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

17. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

18. );

19. 

20. -- 创建索引

21. CREATE INDEX idx_ai_conversations_session ON ai_conversations(session_id);

22. CREATE INDEX idx_ai_conversations_user_mode ON ai_conversations(user_id, mode);

23. CREATE INDEX idx_ai_conversations_created ON ai_conversations(created_at);

24. 

25. -- AI知识库表

26. CREATE TABLE ai_knowledge_base (

27. id BIGSERIAL PRIMARY KEY,

28. question_pattern TEXT NOT NULL,

29. answer_template TEXT NOT NULL,

30. category VARCHAR(100) NOT NULL,

31. tags JSONB,

32. confidence_score DECIMAL(3,2) DEFAULT 1.0,

33. usage_count INTEGER DEFAULT 0,

34. success_rate DECIMAL(4,3) DEFAULT 1.0,

35. mode VARCHAR(20) NOT NULL,

36. enabled BOOLEAN DEFAULT true,

37. created_by BIGINT,

38. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

39. updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

40. );

41. 

42. -- 用户行为分析表

43. CREATE TABLE ai_user_behavior (

44. id BIGSERIAL PRIMARY KEY,

45. user_id BIGINT NOT NULL,

46. behavior_type VARCHAR(50) NOT NULL,

47. behavior_data JSONB NOT NULL,

48. mode VARCHAR(20) NOT NULL,

49. timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP

50. );

51. 

52. -- 情感分析记录表

53. CREATE TABLE ai_sentiment_analysis (

54. id BIGSERIAL PRIMARY KEY,

55. conversation_id BIGINT REFERENCES ai_conversations(id),

56. text_sentiment VARCHAR(20),

57. voice_sentiment VARCHAR(20),

58. combined_sentiment VARCHAR(20),

59. intensity_score DECIMAL(3,2),

60. emotion_tags JSONB,

61. analysis_data JSONB,

62. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

63. );

64. 

65. -- 学习轨迹表

66. CREATE TABLE ai_learning_trajectory (

67. id BIGSERIAL PRIMARY KEY,

68. model_version VARCHAR(50) NOT NULL,

69. learning_type VARCHAR(50) NOT NULL,

70. before_metrics JSONB,

71. after_metrics JSONB,

72. improvement_score DECIMAL(4,3),

73. learning_data JSONB,

74. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

75. );

76. 

77. -- 预测分析表

78. CREATE TABLE ai_predictions (

79. id BIGSERIAL PRIMARY KEY,

80. user_id BIGINT,

81. prediction_type VARCHAR(50) NOT NULL,

82. predicted_actions JSONB,

83. confidence_scores JSONB,

84. context_data JSONB,

85. actual_outcome JSONB,

86. accuracy_score DECIMAL(4,3),

87. created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

88. );

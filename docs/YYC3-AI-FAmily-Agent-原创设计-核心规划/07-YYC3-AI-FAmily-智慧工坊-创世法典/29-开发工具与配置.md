### Docker开发环境

1.  \# 后端服务Dockerfile

2.  FROM openjdk:17-jdk-slim

3.  

4.  \# 安装必要的工具

5.  RUN apt-get update && apt-get install -y \\

6.  curl \\

7.  && rm -rf /var/lib/apt/lists/\*

8.  

9.  \# 创建应用目录

10. WORKDIR /app

11. 

12. \# 复制JAR文件

13. COPY target/smart-restaurant-service.jar app.jar

14. 

15. \# 创建非root用户

16. RUN groupadd -r appuser && useradd -r -g appuser appuser

17. RUN chown -R appuser:appuser /app

18. USER appuser

19. 

20. \# 暴露端口

21. EXPOSE 8080

22. 

23. \# 健康检查

24. HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \\

25. CMD curl -f http://localhost:8080/actuator/health \|\| exit 1

26. 

27. \# 启动应用

28. ENTRYPOINT \["java", "-jar", "app.jar"\]

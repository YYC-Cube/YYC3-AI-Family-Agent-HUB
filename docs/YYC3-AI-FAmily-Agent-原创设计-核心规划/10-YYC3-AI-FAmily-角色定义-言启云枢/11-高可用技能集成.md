## 高可用技能集成

### 1. 技能注册中心

1.  // packages/skills/src/registry/SkillsRegistry.ts

2.  

3.  /\*\*

4.  \* @description 技能注册中心

5.  \* @module skills-registry

6.  \* @version 2.0.0

7.  \*/

8.  

9.  export interface SkillDefinition {

10. id: string;

11. name: string;

12. version: string;

13. category: SkillCategory;

14. description: string;

15. inputSchema: JSONSchema;

16. outputSchema?: JSONSchema;

17. dependencies?: string\[\];

18. provider: 'openai' \| 'ollama' \| 'anthropic' \| 'local';

19. capabilities: string\[\];

20. sla: {

21. availability: number;

22. latency: number;

23. throughput: number;

24. };

25. }

26. 

27. export class SkillsRegistry {

28. private skills: Map\<string, SkillDefinition\> = new Map();

29. private healthChecker: HealthChecker;

30. private loadBalancer: LoadBalancer;

31. 

32. /\*\*

33. \* 注册技能

34. \*/

35. async register(skill: SkillDefinition): Promise\<void\> {

36. // 验证技能定义

37. await this.validateSkill(skill);

38. 

39. // 健康检查

40. const health = await this.healthChecker.check(skill);

41. if (!health.healthy) {

42. throw new Error(\`Skill \${skill.id} health check failed\`);

43. }

44. 

45. this.skills.set(skill.id, skill);

46. 

47. // 更新负载均衡器

48. this.loadBalancer.update(skill);

49. }

50. 

51. /\*\*

52. \* 发现技能

53. \*/

54. async discover(query: SkillQuery): Promise\<SkillDefinition\[\]\> {

55. let results = Array.from(this.skills.values());

56. 

57. // 按类别过滤

58. if (query.category) {

59. results = results.filter(s =\> s.category === query.category);

60. }

61. 

62. // 按能力过滤

63. if (query.capabilities) {

64. results = results.filter(s =\>

65. query.capabilities!.every(c =\> s.capabilities.includes(c))

66. );

67. }

68. 

69. // 按SLA过滤

70. if (query.minAvailability) {

71. results = results.filter(s =\> s.sla.availability \>= query.minAvailability!);

72. }

73. 

74. return results;

75. }

76. 

77. /\*\*

78. \* 执行技能

79. \*/

80. async execute(skillId: string, input: any): Promise\<SkillResult\> {

81. const skill = this.skills.get(skillId);

82. if (!skill) {

83. throw new Error(\`Skill \${skillId} not found\`);

84. }

85. 

86. // 负载均衡选择实例

87. const instance = await this.loadBalancer.select(skill);

88. 

89. // 执行

90. const startTime = Date.now();

91. try {

92. const result = await instance.execute(input);

93. 

94. return {

95. success: true,

96. output: result,

97. metadata: {

98. duration: Date.now() - startTime,

99. instance: instance.id,

100. },

101. };

102. } catch (error) {

103. // 故障转移

104. const fallback = await this.loadBalancer.getFallback(skill);

105. if (fallback) {

106. return await fallback.execute(input);

107. }

108. 

109. throw error;

110. }

111. }

112. }

### 2. 技能组合与编排

1.  // packages/skills/src/orchestration/SkillOrchestrator.ts

2.  

3.  export class SkillOrchestrator {

4.  /\*\*

5.  \* 编排技能链

6.  \*/

7.  async orchestrateChain(config: ChainConfig): Promise\<ChainResult\> {

8.  const results: StepResult\[\] = \[\];

9.  

10. for (const step of config.steps) {

11. const skill = await this.registry.get(step.skillId);

12. 

13. // 准备输入

14. const input = await this.prepareInput(step, results);

15. 

16. // 执行技能

17. const result = await this.registry.execute(step.skillId, input);

18. 

19. // 质量门检查

20. if (step.qualityGate) {

21. const passed = await this.evaluateQualityGate(step.qualityGate, result);

22. if (!passed) {

23. throw new QualityGateError(\`Quality gate failed at step \${step.id}\`);

24. }

25. }

26. 

27. results.push({ step, result });

28. }

29. 

30. return this.synthesizeResults(results);

31. }

32. 

33. /\*\*

34. \* 并行执行技能

35. \*/

36. async orchestrateParallel(config: ParallelConfig): Promise\<ParallelResult\> {

37. const executions = config.skills.map(skill =\>

38. this.registry.execute(skill.id, skill.input)

39. );

40. 

41. const results = await Promise.allSettled(executions);

42. 

43. return {

44. successful: results.filter(r =\> r.status === 'fulfilled').length,

45. failed: results.filter(r =\> r.status === 'rejected').length,

46. results: results.map(r =\> r.status === 'fulfilled' ? r.value : r.reason),

47. };

48. }

49. }

207. \### 3.1 GlobalStoreRegistry (全局注册中心)

208. 

209. \*\*文件位置\*\*: \`src/app/stores/global-store-registry.ts\`

210. 

211. \`\`\`typescript

212. /\*\*

213. \* 全局存储注册中心

214. \*

215. \* 职责:

216. \* 1. 管理所有数据存储的单例实例

217. \* 2. 提供订阅/通知机制

218. \* 3. 实现跨标签页数据同步

219. \* 4. 支持数据导入/导出/重置

220. \*/

221. class GlobalStoreRegistryImpl {

222. private static instance: GlobalStoreRegistryImpl \| null = null;

223. private stores: Map\<StoreKey, LocalStore\<{ id: string }\>\> = new Map();

224. private subscribers: Map\<StoreKey, Set\<StoreChangeCallback\>\> = new Map();

225. private broadcastChannel: BroadcastChannel \| null = null;

226. 

227. // 单例模式

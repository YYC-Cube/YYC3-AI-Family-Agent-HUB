288. async route(userInput: string): Promise\<RoutingResult\> {

291. const routingMap: Record\<string, string\[\]\> = {

292. 'navigation': \['navigator'\],

311. /\*\*

312. \* 协同执行 - 多Agent协作处理复杂任务

313. \*/

314. async collaborate(task: ComplexTask): Promise\<CollaborativeResult\> {

315. const plan = await this.metaOracle.plan(task);

316. 

317. const results: MemberResult\[\] = \[\];

318. 

319. for (const step of plan.steps) {

320. const member = this.members.get(step.member);

321. if (!member) continue;

322. 

323. const result = await this.executeWithRetry(member, step);

324. results.push(result);

325. 

326. if (step.requiresReview) {

327. const review = await this.members.get('master')!.review(result);

328. if (!review.approved) {

329. result = await this.executeWithRetry(member, step);

330. }

331. }

332. 

333. this.executionHistory.push({

334. task: step,

335. member: step.member,

336. result,

337. timestamp: new Date(),

338. });

339. }

340. 

341. return this.synthesize(results);

342. }

343. 

344. /\*\*

345. \* 带重试的执行

346. \*/

347. private async executeWithRetry(

348. member: AIMember,

349. step: ExecutionStep,

350. maxRetries: number = 3

351. ): Promise\<MemberResult\> {

352. let lastError: Error \| null = null;

353. 

354. for (let i = 0; i \< maxRetries; i++) {

355. try {

356. return await member.execute(step.action);

357. } catch (error) {

358. lastError = error;

359. if (this.config.fallbackStrategy === 'switch') {

360. const alternative = await this.findAlternative(step.member);

361. if (alternative) {

362. return await alternative.execute(step.action);

363. }

364. }

365. }

366. }

367. 

368. throw lastError \|\| new Error('Execution failed after retries');

369. }

370. 

371. /\*\*

372. \* 结果综合

373. \*/

374. private synthesize(results: MemberResult\[\]): CollaborativeResult {

375. return {

376. success: results.every(r =\> r.success),

377. outputs: results.map(r =\> r.output),

378. insights: this.extractInsights(results),

379. recommendations: this.generateRecommendations(results),

380. metrics: {

381. totalDuration: results.reduce((sum, r) =\> sum + r.duration, 0),

382. agentCalls: results.length,

383. successRate: results.filter(r =\> r.success).length / results.length,

384. },

385. };

386. }

387. }
